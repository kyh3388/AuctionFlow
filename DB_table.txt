#[DB : auctionflow]

#[KYH_AF_MEMBER]
#회원 관리 테이블
CREATE TABLE KYH_AF_MEMBER (
	M_NO BIGINT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	M_ID VARCHAR(50) NOT NULL UNIQUE,
	M_PW VARCHAR(255) NOT NULL,
	M_NAME VARCHAR(50) NOT NULL,
	M_PHONE_NUMBER VARCHAR(20) NOT NULL,
	M_EMAIL VARCHAR(100) NOT NULL,
	M_ADDRESS VARCHAR(255) NOT NULL,
	CREATED_DATETIME DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
	UPDATED_DATETIME DATETIME NULL
) COMMENT='회원 관리 테이블';

#[KYH_AF_AUCTION]
#경매 상품 관리 테이블
CREATE TABLE KYH_AF_AUCTION (
	A_NO BIGINT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	M_NO BIGINT UNSIGNED NOT NULL,
	A_CATEGORY VARCHAR(20) NOT NULL,
	A_TITLE VARCHAR(200) NOT NULL,
	A_CONTENT TEXT NOT NULL,
	A_START_PRICE BIGINT UNSIGNED NOT NULL,
	A_CURRENT_PRICE BIGINT UNSIGNED NOT NULL,
	HIGHEST_BIDDER_M_NO BIGINT UNSIGNED NULL,
	A_BID_COUNT INT UNSIGNED NOT NULL DEFAULT 0,
	A_STATUS VARCHAR(20) NOT NULL DEFAULT 'ONGOING',
	A_END_DATETIME DATETIME NOT NULL, #기존 경매 종료 시간
	A_CLOSED_DATETIME DATETIME NULL, #실제 경매 종료 시간
	A_PURCHASED_DATETIME DATETIME NULL, #구매 완료 시간
	A_ADMIN_REASON VARCHAR(500) NULL,
	CREATED_DATETIME DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
	UPDATED_DATETIME DATETIME NULL,

	CONSTRAINT FK_KYH_AF_AUCTION_MEMBER
		FOREIGN KEY (M_NO)
		REFERENCES KYH_AF_MEMBER(M_NO),

	CONSTRAINT FK_KYH_AF_AUCTION_HIGHEST_BIDDER
        	FOREIGN KEY (HIGHEST_BIDDER_M_NO)
        	REFERENCES KYH_AF_MEMBER(M_NO)
) COMMENT='경매 상품 관리 테이블';

#[KYH_AF_AUCTION_IMG]
#경매 상품의 이미지 관리 테이블
CREATE TABLE KYH_AF_AUCTION_IMG (
	IMG_NO BIGINT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	A_NO BIGINT UNSIGNED NOT NULL,
	IMG_TYPE VARCHAR(20) NOT NULL,
	IMG_ORIGINAL_NAME VARCHAR(255) NOT NULL,
	IMG_STORED_NAME VARCHAR(255) NOT NULL,
	CREATED_DATETIME DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

	CONSTRAINT FK_KYH_AF_AUCTION_IMG_AUCTION
        	FOREIGN KEY (A_NO)
        	REFERENCES KYH_AF_AUCTION(A_NO)
) COMMENT='경매 상품의 이미지 관리 테이블';

#[KYH_AF_AUCTION_BID]
#경매 중 성공 입찰에 대한 관리를 위한 테이블
CREATE TABLE KYH_AF_AUCTION_BID (
	BID_NO BIGINT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	A_NO BIGINT UNSIGNED NOT NULL,
	M_NO BIGINT UNSIGNED NOT NULL,
	BID_PRICE BIGINT UNSIGNED NOT NULL,
	BID_REQUEST_ID VARCHAR(36) NOT NULL UNIQUE,
	BID_DATETIME DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    	CONSTRAINT FK_KYH_AF_AUCTION_BID_AUCTION
       	FOREIGN KEY (A_NO)
        	REFERENCES KYH_AF_AUCTION(A_NO),

    	CONSTRAINT FK_KYH_AF_AUCTION_BID_MEMBER
        	FOREIGN KEY (M_NO)
        	REFERENCES KYH_AF_MEMBER(M_NO)
) COMMENT='경매 중 성공 입찰에 대한 관리 테이블';
